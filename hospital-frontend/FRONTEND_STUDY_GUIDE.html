4. HOW API CALLS WORK (api.js)

Every single backend call in the ENTIRE app goes through src/utils/api.js
// api.js — the messenger between frontend and backend import axios from 'axios' import { getToken } from './auth' const api = axios.create({ baseURL: 'http://localhost:5253/api', // ← CHANGE SERVER URL HERE }) // Automatically adds JWT token to EVERY request api.interceptors.request.use((config) => { const token = getToken() if (token) config.headers.Authorization = `Bearer ${token}` return config }) // If backend says 401 (not logged in) → auto logout api.interceptors.response.use( (response) => response, (error) => { if (error.response?.status === 401) { localStorage.removeItem('hb_user') window.location.href = '/login' } return Promise.reject(error) } )
How to use api.js in any component:
import api from '../utils/api' const { data } = await api.get('/bills') // GET const { data } = await api.post('/bills', { }) // POST (send data) await api.patch('/bills/1/finalize') // PATCH (update) await api.delete('/bills/1') // DELETE

5. HOW AUTH WORKS (auth.js)

// auth.js — stores user info after login export function setUser(data) { localStorage.setItem('hb_user', JSON.stringify(data)) // stores: { token: "...", role: "Admin", name: "John" } } export function getUser() { return JSON.parse(localStorage.getItem('hb_user')) } export function getToken() { return getUser()?.token // just the JWT token } export function logout() { localStorage.removeItem('hb_user') // clears everything }

6. HOW LOGIN WORKS (Login.jsx)

// Login.jsx — key parts async function handleSubmit(e) { const { data } = await api.post('/auth/login', { email, password }) setUser({ token: data.token, role: data.role, name: data.name }) // Redirects based on role ← CHANGE ROUTES HERE const roleRoutes = { Doctor: '/doctor', LabTech: '/lab', Pharmacist: '/pharmacy', Nurse: '/nurse', Cashier: '/billing', Admin: '/admin', Receptionist: '/reception', } navigate(roleRoutes[data.role]) }

7. THE PATTERN EVERY PAGE FOLLOWS

Every single dashboard in this project follows this EXACT same structure:
export default function SomeDashboard() { // 1. STATE — data the page holds in memory const [bills, setBills] = useState([]) const [loading, setLoading] = useState(false) const [activeTab, setActiveTab] = useState('dashboard') // 2. FETCH ON LOAD — runs when page first opens useEffect(() => { fetchBills() }, []) // 3.
